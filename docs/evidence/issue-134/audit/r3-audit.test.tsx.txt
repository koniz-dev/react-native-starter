import { Text } from 'react-native';
import { act, fireEvent, waitFor } from '@testing-library/react-native';
import { router } from 'expo-router';
import { renderRouter, screen } from 'expo-router/testing-library';
import { authService } from '@/shared/session/authService';
import { defaultUnauthorizedHandler } from '@/shared/session/session';
import { setItem, STORAGE_KEYS } from '@/shared/storage/storage';
import { secureStore } from '@/testing';
import { analyticsSeam } from '@/shared/integrations/analytics';
import { configureIntegrations } from '@/shared/integrations/setup';

const user = { id: 1, email: 'emily@example.com', name: 'Emily Johnson' };
async function storeSession() {
  secureStore.set(STORAGE_KEYS.AUTH_TOKEN, 'stored-token');
  await setItem(STORAGE_KEYS.USER_DATA, user);
}
type R = { name: string; key: string; state?: { index?: number; routes: R[] } };
type App = { getRouterState(): unknown; getPathname(): string };
function dump(app: App) {
  const root = (app.getRouterState() as { routes: R[] }).routes[0]
    ?.state as { index?: number; routes: R[] } | undefined;
  const routes = (root?.routes ?? []).map(r => {
    const inner = r.state;
    const focused =
      inner && inner.index !== undefined ? inner.routes[inner.index]?.name : '';
    return `${r.name}#${r.key.slice(-8)}${focused ? `>${focused}` : ''}`;
  });
  return { routes, index: root?.index, path: app.getPathname(), back: router.canGoBack() };
}
const log = (label: string, v: unknown) =>
  process.stdout.write(`R3 ${label}: ${JSON.stringify(v)}\n`);
function Settings() {
  return <Text>Settings screen</Text>;
}
let boom = false;
function Boom() {
  if (boom) throw new Error('boom');
  return <Text>Boom screen</Text>;
}
async function settle() {
  for (let i = 0; i < 5; i++) {
    await act(async () => {
      jest.advanceTimersByTime?.(50);
    });
  }
}
async function signInOnLogin() {
  await screen.findByText('Welcome Back');
  await fireEvent.changeText(screen.getByTestId('login-username'), 'emily');
  await fireEvent.changeText(screen.getByTestId('login-password'), 'pass');
  await fireEvent.press(screen.getByTestId('login-submit'));
}
const withSettings = (extra: Record<string, unknown> = {}) => ({
  appDir: './app',
  overrides: { settings: Settings, ...extra },
});

let screens: jest.Mock;
beforeEach(() => {
  jest.restoreAllMocks();
  boom = false;
  jest.spyOn(authService, 'fetchProfile').mockResolvedValue(user);
  jest.spyOn(authService, 'login').mockImplementation(async () => {
    await storeSession();
    return { token: 'stored-token', user };
  });
  configureIntegrations();
  screens = jest.fn();
  analyticsSeam.set({ track: jest.fn(), screen: screens, identify: jest.fn() });
});
afterEach(() => analyticsSeam.reset());

describe('cold start', () => {
  it('/profile signed out', async () => {
    const app = await renderRouter('./app', { initialUrl: '/profile' });
    await screen.findByText('Not signed in');
    await settle();
    log('cold /profile out', dump(app));
  });
  it('/login signed in', async () => {
    await storeSession();
    const app = await renderRouter('./app', { initialUrl: '/login' });
    await screen.findByText('Signed in as Emily Johnson');
    await settle();
    log('cold /login in', dump(app));
  });
  it('/no-such then Go home', async () => {
    const app = await renderRouter('./app', { initialUrl: '/no-such' });
    await screen.findByTestId('not-found-home');
    log('cold /no-such', dump(app));
    await fireEvent.press(screen.getByTestId('not-found-home'));
    await settle();
    log('cold /no-such go home', dump(app));
  });
});

describe('error boundaries Go home', () => {
  it.each([
    ['(app)', '/boom', true],
    ['(auth)', '/boom', false],
  ])('%s boundary over Settings', async (group, path, signedIn) => {
    if (signedIn) await storeSession();
    const app = await renderRouter(
      withSettings({ [`${group}/boom`]: Boom }),
      { initialUrl: '/' }
    );
    await screen.findByText(/Not signed in|Signed in as/);
    await act(() => router.push('/settings'));
    await act(() => router.push(path));
    await screen.findByText('Boom screen');
    boom = true;
    await act(() => router.setParams({ x: '1' }));
    await screen.findByTestId('error-go-home');
    log(`${group} boundary before`, dump(app));
    boom = false;
    await fireEvent.press(screen.getByTestId('error-go-home'));
    await settle();
    log(`${group} boundary after`, dump(app));
  });
  it('(tabs) boundary on Explore', async () => {
    const app = await renderRouter(
      withSettings({ '(tabs)/explore': Boom }),
      { initialUrl: '/' }
    );
    await screen.findByText('Not signed in');
    await act(() => router.navigate('/explore'));
    await screen.findByText('Boom screen');
    boom = true;
    await act(() => router.setParams({ x: '1' }));
    await screen.findByTestId('error-go-home');
    log('(tabs) boundary before', dump(app));
    boom = false;
    await fireEvent.press(screen.getByTestId('error-go-home'));
    await settle();
    log('(tabs) boundary after', dump(app));
  });
});

describe('going Home from Settings', () => {
  it.each([
    ['navigate /', () => router.navigate('/')],
    ['dismissTo /', () => router.dismissTo('/')],
    ['push /', () => router.push('/')],
    ['replace /', () => router.replace('/')],
    ['navigate /explore', () => router.navigate('/explore')],
    ['push /explore', () => router.push('/explore')],
    ['dismissTo /explore', () => router.dismissTo('/explore')],
  ])('%s', async (label, go) => {
    const app = await renderRouter(withSettings(), { initialUrl: '/' });
    await screen.findByText('Not signed in');
    await settle();
    await act(() => router.push('/settings'));
    await screen.findByText('Settings screen');
    await settle();
    screens.mockClear();
    await act(() => go());
    log(`from Settings ${label} (sync)`, dump(app));
    await settle();
    log(`from Settings ${label}`, dump(app));
    log(`from Settings ${label} screens`, screens.mock.calls.map(c => c[0]));
  });
  it('push /explore from Profile (signed in)', async () => {
    await storeSession();
    const app = await renderRouter('./app', { initialUrl: '/' });
    await screen.findByText('Signed in as Emily Johnson');
    await act(() => router.push('/profile'));
    await screen.findByText('emily@example.com');
    await act(() => router.push('/explore'));
    await settle();
    log('from Profile push /explore', dump(app));
  });
  it.each([
    ['push /', () => router.push('/')],
    ['push /explore', () => router.push('/explore')],
  ])('from Home %s', async (label, go) => {
    const app = await renderRouter('./app', { initialUrl: '/' });
    await screen.findByText('Not signed in');
    await act(() => go());
    log(`from Home ${label} (sync)`, dump(app));
    await settle();
    log(`from Home ${label}`, dump(app));
  });
});

describe('tracking', () => {
  it('sign in over Settings', async () => {
    const app = await renderRouter(withSettings(), { initialUrl: '/' });
    await screen.findByText('Not signed in');
    await settle();
    await act(() => router.push('/settings'));
    await settle();
    await act(() => router.push('/login'));
    await settle();
    screens.mockClear();
    await signInOnLogin();
    await waitFor(() => expect(screen.queryByText('Welcome Back')).toBeNull());
    await settle();
    log('track sign-in over Settings', { ...dump(app), screens: screens.mock.calls.map(c => c[0]) });
  });
  it('sign out over Settings', async () => {
    await storeSession();
    const app = await renderRouter(withSettings(), { initialUrl: '/' });
    await screen.findByText('Signed in as Emily Johnson');
    await act(() => router.push('/settings'));
    await settle();
    await act(() => router.push('/profile'));
    await screen.findByText('emily@example.com');
    await settle();
    screens.mockClear();
    await fireEvent.press(screen.getByTestId('profile-logout'));
    await settle();
    log('track sign-out over Settings', { ...dump(app), screens: screens.mock.calls.map(c => c[0]) });
  });
  it('expired on Profile over Settings, then Back', async () => {
    await storeSession();
    const app = await renderRouter(withSettings(), { initialUrl: '/' });
    await screen.findByText('Signed in as Emily Johnson');
    await act(() => router.push('/settings'));
    await act(() => router.push('/profile'));
    await screen.findByText('emily@example.com');
    await act(() => defaultUnauthorizedHandler());
    await settle();
    log('expired over Settings', dump(app));
    await act(() => router.back());
    await settle();
    log('expired over Settings then back', dump(app));
  });
  it('sign-in/out from Explore keeps tab', async () => {
    const app = await renderRouter('./app', { initialUrl: '/' });
    await screen.findByText('Not signed in');
    await act(() => router.navigate('/explore'));
    await act(() => router.push('/login'));
    await signInOnLogin();
    await waitFor(() => expect(screen.queryByText('Welcome Back')).toBeNull());
    await settle();
    log('explore sign-in', dump(app));
    await act(() => router.push('/profile'));
    await screen.findByText('emily@example.com');
    await fireEvent.press(screen.getByTestId('profile-logout'));
    await settle();
    log('explore sign-out', dump(app));
  });
});
