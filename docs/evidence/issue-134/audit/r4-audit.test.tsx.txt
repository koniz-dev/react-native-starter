const user = { id: 1, email: 'emily@example.com', name: 'Emily Johnson' };
async function storeSession() {
  secureStore.set(STORAGE_KEYS.AUTH_TOKEN, 'stored-token');
  await setItem(STORAGE_KEYS.USER_DATA, user);
}
type R = { name: string; key: string; state?: { index?: number; routes: R[] } };
type App = { getRouterState(): unknown; getPathname(): string };
function dump(app: App) {
  const root = (app.getRouterState() as { routes: R[] }).routes[0]?.state as { index?: number; routes: R[] } | undefined;
  const routes = (root?.routes ?? []).map(r => {
    const inner = r.state;
    const focused = inner && inner.index !== undefined ? inner.routes.map(x => x.name).join('/') + '@' + inner.routes[inner.index]?.name : '';
    return `${r.name}${focused ? `>${focused}` : ''}`;
  });
  return { routes, index: root?.index, path: app.getPathname(), back: router.canGoBack() };
}

let ctx: SessionContextValue | undefined;

it.each([['/login', false], ['/profile', true]])('J: cold %s then leave group via session', async (url, signedIn) => {
  if (signedIn) await storeSession();
  const app = await renderRouter(ov(), { initialUrl: url });
  await screen.findByText(signedIn ? 'emily@example.com' : 'Welcome Back');
  log(`J cold ${url} before`, dump(app));
  if (signedIn) await fireEvent.press(screen.getByTestId('profile-logout'));
  else {
    await fireEvent.changeText(screen.getByTestId('login-username'), 'emily');
    await fireEvent.changeText(screen.getByTestId('login-password'), 'pass');
    await fireEvent.press(screen.getByTestId('login-submit'));
  }
  await waitFor(() => expect(screen.queryByText(/Welcome Back|emily@example/)).toBeNull());
  await settle();
  log(`J cold ${url} after`, { ...dump(app), errs });
});
