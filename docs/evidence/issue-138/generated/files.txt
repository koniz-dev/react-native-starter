// ===== __tests__/features/account/AccountScreen.test.tsx
import { renderRouter, screen } from 'expo-router/testing-library';
import { setItem, STORAGE_KEYS } from '@/shared/storage/storage';
import { secureStore } from '@/testing';

describe('<AccountScreen />', () => {
  it('opens at /account with its title', async () => {
    // The route needs sign-in: store a session for the app to restore.
    secureStore.set(STORAGE_KEYS.AUTH_TOKEN, 'stored-token');
    await setItem(STORAGE_KEYS.USER_DATA, {
      id: 1,
      email: 'user@example.com',
      name: 'User',
    });
    const app = await renderRouter('./app', { initialUrl: '/account' });

    expect(await screen.findByTestId('account-title')).toHaveTextContent(
      'Account'
    );
    expect(app.getPathname()).toBe('/account');
  });
});
